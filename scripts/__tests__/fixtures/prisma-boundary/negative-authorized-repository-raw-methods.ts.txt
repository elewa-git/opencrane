import type { Prisma } from "@prisma/client";
import type { WidgetRepository } from "./widget.types.js";

export class PrismaWidgetRepository implements WidgetRepository
{
	private readonly transaction: Prisma.TransactionClient;

	constructor(transaction: Prisma.TransactionClient)
	{
		this.transaction = transaction;
	}

	async find(): Promise<unknown>
	{
		const database = this.transaction;
		await database.$queryRaw`SELECT 1`;
		await database.$queryRawUnsafe("SELECT 1");
		await database.$executeRaw`SELECT 1`;
		await database.$executeRawUnsafe("SELECT 1");
		return this.transaction.widget.findFirst({});
	}
}
